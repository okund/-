-- =====================================================
-- 校园考勤助手 数据库初始化脚本（可重复执行）
-- 数据库: edu_check
-- 技术栈: MySQL 8.0+ / Spring Boot 3.2 / MyBatis-Plus
-- 说明: 执行时会先删旧表再建新表，每次都是全新数据
-- =====================================================



-- =====================================================
-- 1. 用户表
-- =====================================================
DROP TABLE IF EXISTS `user`;
CREATE TABLE `user` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    `name` VARCHAR(50) NOT NULL COMMENT '姓名',
    `student_id` VARCHAR(20) DEFAULT NULL COMMENT '学号',
    `password` VARCHAR(255) NOT NULL DEFAULT '123456' COMMENT '密码',
    `college` VARCHAR(100) DEFAULT NULL COMMENT '学院',
    `major` VARCHAR(100) DEFAULT NULL COMMENT '专业',
    `grade` VARCHAR(20) DEFAULT NULL COMMENT '年级',
    `avatar` VARCHAR(255) DEFAULT NULL COMMENT '头像URL',
    `phone` VARCHAR(20) DEFAULT NULL COMMENT '手机号',
    `email` VARCHAR(100) DEFAULT NULL COMMENT '邮箱',
    `role` VARCHAR(20) NOT NULL DEFAULT 'student' COMMENT '角色: student/teacher/admin',
    `status` TINYINT NOT NULL DEFAULT 1 COMMENT '状态: 1正常 0禁用',
    `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    `updated_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    PRIMARY KEY (`id`),
    UNIQUE KEY `uk_student_id` (`student_id`),
    INDEX `idx_college` (`college`),
    INDEX `idx_role` (`role`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户表';

-- =====================================================
-- 4. 课程表
-- =====================================================
DROP TABLE IF EXISTS `course`;
CREATE TABLE `course` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    `name` VARCHAR(100) NOT NULL COMMENT '课程名称',
    `teacher` VARCHAR(50) DEFAULT NULL COMMENT '授课教师',
    `location` VARCHAR(200) DEFAULT NULL COMMENT '上课地点',
    `start_time` VARCHAR(20) DEFAULT NULL COMMENT '上课时间 HH:mm',
    `end_time` VARCHAR(20) DEFAULT NULL COMMENT '下课时间 HH:mm',
    `section` VARCHAR(20) DEFAULT NULL COMMENT '节次',
    `week_day` VARCHAR(10) DEFAULT NULL COMMENT '星期几',
    `week_start` INT DEFAULT NULL COMMENT '起始周',
    `week_end` INT DEFAULT NULL COMMENT '结束周',
    `teacher_id` BIGINT DEFAULT NULL COMMENT '教师用户ID',
    `status` TINYINT NOT NULL DEFAULT 1 COMMENT '状态: 1启用 0停用',
    `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    `updated_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    PRIMARY KEY (`id`),
    INDEX `idx_teacher_id` (`teacher_id`),
    INDEX `idx_week_day` (`week_day`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='课程表';

-- =====================================================
-- 5. 上课签到表
-- =====================================================
DROP TABLE IF EXISTS `class_checkin`;
CREATE TABLE `class_checkin` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    `user_id` BIGINT NOT NULL COMMENT '用户ID',
    `course_id` BIGINT DEFAULT NULL COMMENT '课程ID',
    `date` DATE NOT NULL COMMENT '签到日期',
    `checkin_time` DATETIME DEFAULT NULL COMMENT '签到时间',
    `method` VARCHAR(20) DEFAULT NULL COMMENT '签到方式: location/qrcode',
    `location_lat` DECIMAL(10,6) DEFAULT NULL COMMENT '纬度',
    `location_lng` DECIMAL(10,6) DEFAULT NULL COMMENT '经度',
    `location_addr` VARCHAR(255) DEFAULT NULL COMMENT '定位地址',
    `dynamic_code` VARCHAR(10) DEFAULT NULL COMMENT '动态码',
    `status` VARCHAR(20) NOT NULL DEFAULT 'present' COMMENT '状态: present/absent/late',
    `session_id` VARCHAR(50) DEFAULT NULL COMMENT '签到会话ID',
    `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    PRIMARY KEY (`id`),
    INDEX `idx_user_id` (`user_id`),
    INDEX `idx_course_id` (`course_id`),
    INDEX `idx_date` (`date`),
    INDEX `idx_session_id` (`session_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='上课签到表';

-- =====================================================
-- 6. 查寝打卡表
-- =====================================================
DROP TABLE IF EXISTS `dorm_checkin`;
CREATE TABLE `dorm_checkin` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    `user_id` BIGINT NOT NULL COMMENT '用户ID',
    `date` DATE NOT NULL COMMENT '打卡日期',
    `checkin_time` DATETIME DEFAULT NULL COMMENT '打卡时间',
    `location_lat` DECIMAL(10,6) DEFAULT NULL COMMENT '纬度',
    `location_lng` DECIMAL(10,6) DEFAULT NULL COMMENT '经度',
    `location_addr` VARCHAR(255) DEFAULT NULL COMMENT '定位地址',
    `building` VARCHAR(100) DEFAULT NULL COMMENT '宿舍楼',
    `room` VARCHAR(50) DEFAULT NULL COMMENT '宿舍号',
    `in_dorm_area` TINYINT NOT NULL DEFAULT 0 COMMENT '是否在宿舍区域',
    `face_verified` TINYINT NOT NULL DEFAULT 0 COMMENT '人脸是否验证通过',
    `face_image_url` VARCHAR(255) DEFAULT NULL COMMENT '人脸图片URL',
    `status` VARCHAR(20) NOT NULL DEFAULT 'normal' COMMENT '状态: normal/late',
    `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    PRIMARY KEY (`id`),
    INDEX `idx_user_id` (`user_id`),
    INDEX `idx_date` (`date`),
    INDEX `idx_status` (`status`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='查寝打卡表';

-- =====================================================
-- 7. 实习表
-- =====================================================
DROP TABLE IF EXISTS `internship`;
CREATE TABLE `internship` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    `user_id` BIGINT NOT NULL COMMENT '用户ID',
    `company` VARCHAR(100) DEFAULT NULL COMMENT '公司名称',
    `role` VARCHAR(100) DEFAULT NULL COMMENT '实习岗位',
    `start_date` DATE DEFAULT NULL COMMENT '实习开始日期',
    `end_date` DATE DEFAULT NULL COMMENT '实习结束日期',
    `total_days` INT DEFAULT NULL COMMENT '总实习天数',
    `completed_days` INT NOT NULL DEFAULT 0 COMMENT '已完成天数',
    `status` VARCHAR(20) NOT NULL DEFAULT 'active' COMMENT '状态: active/completed',
    `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    `updated_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    PRIMARY KEY (`id`),
    INDEX `idx_user_id` (`user_id`),
    INDEX `idx_status` (`status`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='实习表';

-- =====================================================
-- 8. 实习打卡表
-- =====================================================
DROP TABLE IF EXISTS `intern_checkin`;
CREATE TABLE `intern_checkin` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    `user_id` BIGINT NOT NULL COMMENT '用户ID',
    `internship_id` BIGINT DEFAULT NULL COMMENT '实习ID',
    `date` DATE NOT NULL COMMENT '打卡日期',
    `checkin_time` DATETIME DEFAULT NULL COMMENT '打卡时间',
    `day_number` INT DEFAULT NULL COMMENT '第N天打卡',
    `location_lat` DECIMAL(10,6) DEFAULT NULL COMMENT '纬度',
    `location_lng` DECIMAL(10,6) DEFAULT NULL COMMENT '经度',
    `location_addr` VARCHAR(255) DEFAULT NULL COMMENT '定位地址',
    `in_company_area` TINYINT NOT NULL DEFAULT 0 COMMENT '是否在公司区域',
    `log_content` TEXT DEFAULT NULL COMMENT '工作日志',
    `status` VARCHAR(20) NOT NULL DEFAULT 'completed' COMMENT '状态: completed',
    `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    PRIMARY KEY (`id`),
    INDEX `idx_user_id` (`user_id`),
    INDEX `idx_internship_id` (`internship_id`),
    INDEX `idx_date` (`date`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='实习打卡表';

-- =====================================================
-- 9. 实习图片表
-- =====================================================
DROP TABLE IF EXISTS `intern_photo`;
CREATE TABLE `intern_photo` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    `checkin_id` BIGINT NOT NULL COMMENT '实习打卡ID',
    `url` VARCHAR(255) NOT NULL COMMENT '图片URL',
    `sort_order` INT NOT NULL DEFAULT 0 COMMENT '排序',
    `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    PRIMARY KEY (`id`),
    INDEX `idx_checkin_id` (`checkin_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='实习图片表';

-- =====================================================
-- 10. 打卡统计表
-- =====================================================
DROP TABLE IF EXISTS `checkin_stats`;
CREATE TABLE `checkin_stats` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    `user_id` BIGINT NOT NULL COMMENT '用户ID',
    `dorm_total` INT NOT NULL DEFAULT 0 COMMENT '查寝打卡总数',
    `class_total` INT NOT NULL DEFAULT 0 COMMENT '上课签到总数',
    `intern_total` INT NOT NULL DEFAULT 0 COMMENT '实习打卡总数',
    `streak_days` INT NOT NULL DEFAULT 0 COMMENT '连续打卡天数',
    `total_points` INT NOT NULL DEFAULT 0 COMMENT '总积分',
    `updated_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    PRIMARY KEY (`id`),
    UNIQUE KEY `uk_user_id` (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='打卡统计表';

-- =====================================================
-- 11. 用户人脸表
-- =====================================================
DROP TABLE IF EXISTS `user_face`;
CREATE TABLE `user_face` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    `user_id` BIGINT NOT NULL COMMENT '用户ID',
    `face_url` VARCHAR(255) DEFAULT NULL COMMENT '人脸图片存储路径',
    `face_token` VARCHAR(255) DEFAULT NULL COMMENT '百度云返回的face_token',
    `registered` TINYINT NOT NULL DEFAULT 0 COMMENT '是否已注册人脸: 0否 1是',
    `version` INT NOT NULL DEFAULT 0 COMMENT '注册次数（每次更新+1）',
    `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    `updated_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    PRIMARY KEY (`id`),
    UNIQUE KEY `uk_user_id` (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户人脸表';
